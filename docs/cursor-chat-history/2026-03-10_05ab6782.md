  View,

type Instrument = {
  symbol: string;
  name: string;
};

type Holding = {
  id: string;
  market: MarketType;
  symbol: string;
  name: string;
  quantity: number;
  price: number;
};

// Şimdilik örnek bir kaç enstrüman – sonra listeyi büyütebiliriz
const BIST_INSTRUMENTS: Instrument[] = [
  { symbol: 'AKBNK', name: 'Akbank' },
  { symbol: 'GARAN', name: 'Garanti BBVA' },
  { symbol: 'THYAO', name: 'Türk Hava Yolları' },
];

const COMMODITY_INSTRUMENTS: Instrument[] = [
  { symbol: 'XAUTRY', name: 'Altın (Gram)' },
  { symbol: 'XAGTRY', name: 'Gümüş (Gram)' },
  { symbol: 'XPDTRY', name: 'Paladyum' },
];

// Geçici “anlık fiyat” tablosu
const PRICE_MAP: Record<string, number> = {
  AKBNK: 50.2,
  GARAN: 63.4,
  THYAO: 310.5,
  XAUTRY: 2500,
  XAGTRY: 30,
  XPDTRY: 900,
};

export default function HomeScreen() {
  const [selectedMarket, setSelectedMarket] = useState<MarketType>('bist');
  const [holdings, setHoldings] = useState<Holding[]>([]);
  const [addModalVisible, setAddModalVisible] = useState(false);
  const [newMarket, setNewMarket] = useState<MarketType>('bist');
  const [selectedInstrument, setSelectedInstrument] = useState<Instrument | null>(null);
  const [quantityInput, setQuantityInput] = useState<string>('');

  const instruments = newMarket === 'bist' ? BIST_INSTRUMENTS : COMMODITY_INSTRUMENTS;

  const filteredHoldings = holdings.filter((h) => h.market === selectedMarket);

  const totalValue = filteredHoldings.reduce((sum, h) => sum + h.quantity * h.price, 0);

  const handleAddHolding = () => {
    if (!selectedInstrument) return;
    const quantity = Number(quantityInput.replace(',', '.'));
    if (!quantity || quantity <= 0) return;

    const price = PRICE_MAP[selectedInstrument.symbol] ?? 0;

    const newHolding: Holding = {
      id: `${selectedInstrument.symbol}-${Date.now()}`,
      market: newMarket,
      symbol: selectedInstrument.symbol,
      name: selectedInstrument.name,
      quantity,
      price,
    };

    setHoldings((prev) => [...prev, newHolding]);
    setAddModalVisible(false);
    setSelectedInstrument(null);
    setQuantityInput('');
    setNewMarket(selectedMarket);
  };

  const renderHoldingItem = ({ item }: { item: Holding }) => {
    const value = item.quantity * item.price;
    return (
      <ThemedView style={styles.holdingCard}>
        <View style={styles.holdingHeader}>
          <ThemedText type="defaultSemiBold">{item.symbol}</ThemedText>
          <ThemedText>{item.name}</ThemedText>
        </View>
        <View style={styles.holdingRow}>
          <ThemedText>Adet: {item.quantity}</ThemedText>
          <ThemedText>Fiyat: {item.price.toFixed(2)} ₺</ThemedText>
          <ThemedText type="defaultSemiBold">Tutar: {value.toFixed(2)} ₺</ThemedText>
        </View>
      </ThemedView>
    );
  };

    <SafeAreaView style={styles.container}>
      <ThemedView style={styles.header}>
        <ThemedText type="title">Portföyüm</ThemedText>
      </ThemedView>

      {/* Üstte iki ana sekme: Borsa İstanbul / Emtia */}
      <ThemedView style={styles.marketTabs}>
        <TouchableOpacity
          style={[styles.marketTab, selectedMarket === 'bist' && styles.marketTabActive]}
          onPress={() => setSelectedMarket('bist')}>
          <ThemedText type="defaultSemiBold">Borsa İstanbul</ThemedText>
        </TouchableOpacity>
        <TouchableOpacity
          style={[styles.marketTab, selectedMarket === 'commodity' && styles.marketTabActive]}
          onPress={() => setSelectedMarket('commodity')}>
          <ThemedText type="defaultSemiBold">Emtia</ThemedText>
        </TouchableOpacity>
      </ThemedView>

      {/* Seçili sekmenin toplam değeri */}
      <ThemedView style={styles.summaryCard}>
        <ThemedText type="subtitle">
          {selectedMarket === 'bist' ? 'Borsa İstanbul Toplam Değer' : 'Emtia Toplam Değer'}
        </ThemedText>
        <ThemedText type="title">{totalValue.toFixed(2)} ₺</ThemedText>
      </ThemedView>

      {/* Altında enstrüman listesi (envanterim) */}
      <FlatList
        data={filteredHoldings}
        keyExtractor={(item) => item.id}
        contentContainerStyle={styles.listContent}
        ListEmptyComponent={
          <ThemedText>

              : 'Henüz emtia için kayıt yok.'}
          </ThemedText>
        }
        renderItem={renderHoldingItem}
      />

      {/* Büyük + butonu */}
      <TouchableOpacity
        style={styles.fab}
        onPress={() => {
          setNewMarket(selectedMarket);
          setAddModalVisible(true);
        }}>
        <ThemedText type="title">+</ThemedText>
      </TouchableOpacity>

      {/* Kayıt ekleme penceresi */}
      <Modal
        visible={addModalVisible}
        animationType="slide"
        transparent
        onRequestClose={() => setAddModalVisible(false)}>
        <View style={styles.modalOverlay}>
          <ThemedView style={styles.modalContent}>
            <ThemedText type="subtitle">Yeni Kayıt Ekle</ThemedText>

            {/* Borsa İstanbul / Emtia seçimi */}
            <View style={styles.modalMarketTabs}>
              <TouchableOpacity
                style={[
                  styles.modalMarketTab,
                  newMarket === 'bist' && styles.modalMarketTabActive,
                ]}
                onPress={() => {
                  setNewMarket('bist');
                  setSelectedInstrument(null);
                }}>
                <ThemedText>Borsa İstanbul</ThemedText>
              </TouchableOpacity>
              <TouchableOpacity
                style={[
                  styles.modalMarketTab,
                  newMarket === 'commodity' && styles.modalMarketTabActive,
                ]}
                onPress={() => {
                  setNewMarket('commodity');
                  setSelectedInstrument(null);
                }}>
                <ThemedText>Emtia</ThemedText>
              </TouchableOpacity>
            </View>

            {/* Enstrüman listesi */}
            <ThemedText style={styles.sectionTitle}>
              {newMarket === 'bist' ? 'Hisse Seç' : 'Emtia Seç'}
            </ThemedText>

            <FlatList
              data={instruments}
              keyExtractor={(item) => item.symbol}
              style={styles.instrumentList}
              renderItem={({ item }) => {
                const isSelected = selectedInstrument?.symbol === item.symbol;
                return (
                  <TouchableOpacity
                    style={[
                      styles.instrumentItem,
                      isSelected && styles.instrumentItemSelected,
                    ]}
                    onPress={() => setSelectedInstrument(item)}>
                    <ThemedText type="defaultSemiBold">{item.symbol}</ThemedText>
                    <ThemedText>{item.name}</ThemedText>
                  </TouchableOpacity>

## ASSISTANT

Bu hata, Windows’un `git` komutunu bulamadığı anlamına geliyor. Şunları kontrol et:

---

### 1. Git gerçekten kurulu mu?

- **Başlat** menüsünde “Git” veya “Git Bash” ara.
- Veya **Dosya Gezgini** ile şuna bak:  
  `C:\Program Files\Git\bin\git.exe`  
  Bu dosya yoksa Git hiç kurulmamış demektir → [git-scm.com/download/win](https://git-scm.com/download/win) adresinden indirip kur, kurulumda PATH seçeneğini işaretle.

---

### 2. Kuruluysa: Cursor’u tamamen kapatıp aç

- Sadece terminali kapatmak yetmez; **Cursor’u kapatıp yeniden aç**.
- Sonra yeni bir terminal açıp tekrar dene:

```powershell
cd "C:\Users\Haşim Öztürk\OneDrive\Desktop\CURSOR\portfoy-takip"
git init
```

---

### 3. Hâlâ “git is not recognized” diyorsa: Tam yolu kullan

Git kurulu ama PATH’e eklenmemiş olabilir. O zaman şunu dene:

```powershell
& "C:\Program Files\Git\bin\git.exe" init
& "C:\Program Files\Git\bin\git.exe" add .
& "C:\Program Files\Git\bin\git.exe" commit -m "Ilk surum: portfoy takip uygulamasi"
```

(Eğer Git farklı bir dizine kurulduysa —ör. `C:\Program Files (x86)\Git\bin\git.exe— o yolu kullan.)

---

Özet: Önce Git’in kurulu olduğundan emin ol, Cursor’u kapat-aç et, sonra `git init` veya tam yollu komutu çalıştır. Hâlâ aynı hatayı alırsan, “Git kurulumu yaptım / yapmadım” ve tam hata çıktısını yaz.
